  <script>
    let char;
    const historyDiv = document.getElementById('history');
    let lastValue = "";
    let lastTime = 0;

    async function connect() {
      try {
        const device = await navigator.bluetooth.requestDevice({
          filters: [{ services: ['965faa66-e492-4af1-8c42-deef6500ee8c'] }]
        });
        const server = await device.gatt.connect();
        const service = await server.getPrimaryService('####################################');
        char = await service.getCharacteristic('$$$$$$$$$$$$$$$$$$$$$$$$$$$$$$$$$$$$');
        await char.startNotifications();
        char.addEventListener('characteristicvaluechanged', onValueChanged);
        addLog("--- 接続成功 ---");
      } catch (e) {
        alert("接続失敗: " + e);
      }
    }

    // 文字列（Kp=1.0など）を送信
    async function sendText() {
      const val = document.getElementById('val').value;
      await transmit(val);
    }

    // 操縦コマンド（f, b, r, l）を送信
    async function sendCmd(cmd) {
      await transmit(cmd);
    }

    // 共通の送信処理
    async function transmit(data) {
      if (!char) { alert("接続されていません"); return; }
      try {
        await char.writeValue(new TextEncoder().encode(data));
      } catch (e) {
        addLog("送信失敗: " + e);
      }
    }

    function onValueChanged(event) {
      const value = new TextDecoder().decode(event.target.value);
      const now = Date.now();
      if (value === lastValue && (now - lastTime) < 100) return;
      lastValue = value;
      lastTime = now;
      addLog(value);
    }

    function addLog(text) {
      const now = new Date();
      const timeStr = now.getHours().toString().padStart(2, '0') + ":" + 
                      now.getMinutes().toString().padStart(2, '0') + ":" + 
                      now.getSeconds().toString().padStart(2, '0');
      const item = document.createElement('div');
      item.className = 'log-item';
      item.innerHTML = `<span class="val-group">${text}</span><span class="time">[${timeStr}]</span>`;
      historyDiv.appendChild(item);
      historyDiv.scrollTop = historyDiv.scrollHeight;
    }
  </script>
